⚡ Common Usage

using Cache.Adapters.Memory;
using Cache.Core;

// 1. Create — 30 s default TTL, sweep expired entries every 10 s
using var cache = new TtlCache<string>(
    defaultTtl:    TimeSpan.FromSeconds(30),
    sweepInterval: TimeSpan.FromSeconds(10));

// 2. Write — uses default TTL
cache.Set("user:42", "Alice");

// 3. Write — custom absolute TTL
cache.Set("session:abc", "token",
    new CacheEntryOptions(AbsoluteTtl: TimeSpan.FromMinutes(5)));

// 4. Write — sliding TTL capped by an absolute ceiling
cache.Set("feed:home", "data",
    new CacheEntryOptions(
        AbsoluteTtl: TimeSpan.FromMinutes(10),
        SlidingTtl:  TimeSpan.FromMinutes(2)));

// 5. Read
if (cache.TryGet("user:42", out var user))
    Console.WriteLine(user);   // "Alice"

// 6. Delete
cache.Remove("user:42");

// 7. Metrics
CacheMetricsSnapshot snap = cache.GetMetricsSnapshot();
Console.WriteLine($"Hit rate: {snap.HitRate:P1}  Entries: {snap.Count}");

// 8. Manual sweep (normally automatic)
int evicted = cache.SweepExpiredEntries();

// Dispose stops the background sweep timer
